#!/usr/bin/env python3
"""
Cleanup Review — a cautious Linux file-cleanup assistant.

Scans the user's home directory, identifies review candidates, and moves
selected files into ~/CleanupReview. Nothing is permanently deleted unless
the user explicitly runs the purge command.

Examples:
    python3 cleanup.py scan
    python3 cleanup.py review
    python3 cleanup.py restore
    python3 cleanup.py purge
"""

import argparse
import hashlib
import json
import os
import shutil
import sys
import time
from collections import defaultdict
from pathlib import Path

HOME = Path.home()
REVIEW = HOME / "CleanupReview"
MANIFEST = REVIEW / "manifest.json"

CATEGORIES = ("temporary", "old_logs", "screenshots", "pdfs", "duplicates")

# These directories are deliberately not scanned for automatic moving.
SKIP_DIRS = {
    ".cache", ".local", ".config", ".mozilla", ".var",
    ".thunderbird", ".steam", ".wine", "node_modules",
    ".git", "__pycache__", "CleanupReview",
}

TEMP_SUFFIXES = {
    ".tmp", ".temp", ".part", ".partial", ".swp", ".swo",
    ".bak", ".old", ".cache",
}
LOG_SUFFIXES = {".log"}
SCREENSHOT_WORDS = ("screenshot", "screen shot", "スクリーンショット")
SCREENSHOT_DIRS = {"screenshots", "screen-shots", "screen_shots"}

# Cache and Trash are reported, never moved automatically.
REPORTED_ONLY = {
    "cache": HOME / ".cache",
    "trash": HOME / ".local" / "share" / "Trash",
}


def human(size):
    size = float(size)
    for unit in ("B", "KB", "MB", "GB", "TB"):
        if size < 1024 or unit == "TB":
            return f"{size:.1f} {unit}"
        size /= 1024
    return f"{size:.1f} TB"


def file_hash(path):
    """Return a SHA-256 digest, or None if the file cannot be read."""
    digest = hashlib.sha256()
    try:
        with path.open("rb") as stream:
            for chunk in iter(lambda: stream.read(1024 * 1024), b""):
                digest.update(chunk)
        return digest.hexdigest()
    except OSError:
        return None


def load_manifest():
    if not MANIFEST.exists():
        return {"version": 1, "items": []}
    try:
        data = json.loads(MANIFEST.read_text(encoding="utf-8"))
        if data.get("version") != 1 or not isinstance(data.get("items"), list):
            raise ValueError("Unsupported manifest format")
        return data
    except (OSError, json.JSONDecodeError, ValueError) as exc:
        raise RuntimeError(
            f"Cannot read {MANIFEST}: {exc}\n"
            "Do not overwrite the manifest; inspect or back it up first."
        ) from exc


def save_manifest(data):
    REVIEW.mkdir(parents=True, exist_ok=True)
    temporary_manifest = MANIFEST.with_suffix(".json.tmp")
    temporary_manifest.write_text(
        json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    os.replace(temporary_manifest, MANIFEST)


def is_inside(path, directory):
    try:
        path.resolve().relative_to(directory.resolve())
        return True
    except (ValueError, OSError):
        return False


def classify(path):
    """Return a review category, or None when the file is not a candidate."""
    suffix = path.suffix.lower()
    name = path.name.lower()
    parent_names = {part.lower() for part in path.parts}

    if suffix in TEMP_SUFFIXES or name.startswith("~$"):
        return "temporary"
    if suffix in LOG_SUFFIXES:
        return "old_logs"
    if suffix == ".pdf":
        return "pdfs"
    if suffix in {".png", ".jpg", ".jpeg", ".webp"}:
        if any(word in name for word in SCREENSHOT_WORDS):
            return "screenshots"
        if parent_names & SCREENSHOT_DIRS:
            return "screenshots"
    return None


def scan_files():
    """Walk HOME without following symlinks or entering excluded directories."""
    for root, dirs, filenames in os.walk(HOME, topdown=True, followlinks=False):
        root_path = Path(root)

        dirs[:] = [
            dirname for dirname in dirs
            if dirname not in SKIP_DIRS
            and not (root_path / dirname).is_symlink()
            and not is_inside(root_path / dirname, REVIEW)
        ]

        for filename in filenames:
            path = root_path / filename
            try:
                if path.is_symlink() or not path.is_file():
                    continue
                # Avoid files that change while the scan is running.
                stat = path.stat()
                yield path, stat.st_size, stat.st_mtime
            except OSError:
                continue


def scan():
    print("Scanning your home directory...")
    candidates = []
    hash_groups = defaultdict(list)

    for path, size, modified in scan_files():
        category = classify(path)
        if category:
            candidates.append({
                "path": str(path),
                "category": category,
                "size": size,
                "modified": modified,
            })

        # Hashing every file can be expensive. Only hash files up to 100 MB.
        if size <= 100 * 1024 * 1024 and size > 0:
            digest = file_hash(path)
            if digest:
                hash_groups[digest].append((path, size, modified))

    duplicate_groups = [
        group for group in hash_groups.values() if len(group) > 1
    ]
    duplicate_paths = {
        str(path)
        for group in duplicate_groups
        for path, _, _ in group[1:]
    }

    # Mark extra copies as duplicate candidates. Keep the first copy untouched.
    existing = {item["path"] for item in candidates}
    for group in duplicate_groups:
        for path, size, modified in group[1:]:
            if str(path) not in existing:
                candidates.append({
                    "path": str(path),
                    "category": "duplicates",
                    "size": size,
                    "modified": modified,
                })

    candidates.sort(key=lambda item: (item["category"], item["path"]))
    print(f"\nFound {len(candidates)} review candidates.")
    totals = defaultdict(lambda: [0, 0])
    for item in candidates:
        totals[item["category"]][0] += 1
        totals[item["category"]][1] += item["size"]

    for category in CATEGORIES:
        count, size = totals[category]
        if count:
            print(f"  {category:14} {count:5} files   {human(size)}")

    print("\nReported-only locations:")
    for label, folder in REPORTED_ONLY.item s():
        try:
            if folder.exists():
                print(f"  {label:10} {folder}   (not moved)")
            else:
                print(f"  {label:10} not present")
        except OSError:
            print(f"  {label:10} could not be checked")

    if not candidates:
        return

    print("\nChoose candidates to move into the review folder.")
    print("Enter: all | category name | comma-separated numbers | none")
    for number, item in enumerate(candidates, 1):
        print(
            f"{number:4}. [{item['category']:12}] "
            f"{human(item['size']):>10}  {item['path']}"
        )

    choice = input("\nSelection: ").strip().lower()
    if choice in ("", "none", "n"):
        print("No files moved.")
        return

    if choice == "all":
        selected = candidates
    elif choice in CATEGORIES:
        selected = [item for item in candidates if item["category"] == choice]
    else:
        try:
            numbers = {int(value.strip()) for value in choice.split(",")}
            if any(number < 1 or number > len(candidates) for number in numbers):
                raise ValueError
            selected = [candidates[number - 1] for number in sorted(numbers)]
        except ValueError:
            print("Invalid selection. Nothing was moved.")
            return

    confirm = input(
        f"Move {len(selected)} file(s) to {REVIEW}? [y/N]: "
    ).strip().lower()
    if confirm != "y":
        print("Cancelled.")
        return

    stage_files(selected)


def unique_destination(destination):
    """Avoid overwriting an existing review file."""
    if not destination.exists():
        return destination
    counter = 1
    while True:
        candidate = destination.with_name(
            f"{destination.stem}__{counter}{destination.suffix}"
        )
        if not candidate.exists():
            return candidate
        counter += 1


def stage_files(items):
    REVIEW.mkdir(parents=True, exist_ok=True)
    manifest = load_manifest()
    moved = 0

    for item in items:
        source = Path(item["path"])
        try:
            # Re-check before moving: the scan result may now be stale.
            if not source.is_file() or source.is_symlink():
                print(f"Skipped (missing or unsafe): {source}")
                continue
            if not is_inside(source, HOME) or is_inside(source, REVIEW):
                print(f"Skipped (outside home/review): {source}")
                continue

            relative = source.relative_to(HOME)
            destination = unique_destination(REVIEW / item["category"] / relative)
            destination.parent.mkdir(parents=True, exist_ok=True)

            # shutil.move may cross filesystems; it handles that case.
            shutil.move(str(source), str(destination))
            manifest["items"].append({
                "id": len(manifest["items"]) + 1,
                "category": item["category"],
                "original": str(source),
                "stored": str(destination),
                "size": item["size"],
                "staged_at": time.time(),
                "status": "staged",
            })
            moved += 1
            print(f"Moved: {source.name}")
        except OSError as exc:
            print(f"Could not move {source}: {exc}")

    save_manifest(manifest)
    print(f"\nMoved {moved} file(s). Originals can be restored with: python3 {Path(sys.argv[0]).name} restore")


def review():
    manifest = load_manifest()
    active = [item for item in manifest["items"] if item["status"] == "staged"]
    if not active:
        print("There are no staged files to review.")
        return

    print(f"{'ID':>4}  {'CATEGORY':14} {'SIZE':>10}  FILE")
    print("-" * 85)
    for item in active:
        print(
            f"{item['id']:4}  {item['category']:14} "
            f"{human(item['size']):>10}  {item['stored']}"
        )
    print(f"\n{len(active)} file(s) awaiting your decision.")
    print(f"Review folder: {REVIEW}")


def restore():
    manifest = load_manifest()
    active = [item for item in manifest["items"] if item["status"] == "staged"]
    if not active:
        print("There are no staged files to restore.")
        return

    print("Files available to restore:")
    for item in active:
        print(f"  {item['id']}: {item['stored']}")

    choice = input("\nRestore IDs (comma-separated) or 'all': ").strip().lower()
    if choice == "all":
        selected = active
    else:
        try:
            ids = {int(value.strip()) for value in choice.split(",")}
            selected = [item for item in active if item["id"] in ids]
            if not selected or len(selected) != len(ids):
                raise ValueError
        except ValueError:
            print("Invalid selection. Nothing restored.")
            return

    confirm = input(f"Restore {len(selected)} file(s)? [y/N]: ").strip().lower()
    if confirm != "y":
        print("Cancelled.")
        return

    for item in selected:
        stored = Path(item["stored"])
        original = Path(item["original"])
        try:
            if not stored.is_file():
                print(f"Missing from review folder: {stored}")
                continue
            if not is_inside(stored, REVIEW) or not is_inside(original, HOME):
                print(f"Skipped unsafe path: {stored}")
                continue
            if original.exists():
                print(f"Destination already exists; not overwriting: {original}")
                continue
            original.parent.mkdir(parents=True, exist_ok=True)
            shutil.move(str(stored), str(original))
            item["status"] = "restored"
            print(f"Restored: {original}")
        except OSError as exc:
            print(f"Could not restore {stored}: {exc}")

    save_manifest(manifest)


def purge():
    manifest = load_manifest()
    active = [item for item in manifest["items"] if item["status"] == "staged"]
    if not active:
        print("There are no staged files to delete.")
        return

    print("This permanently deletes files from the review folder.")
    print("Check the files first with: python3 cleanup.py review")
    for item in active:
        print(f"  {item['id']}: {item['stored']}")

    choice = input("\nDelete IDs (comma-separated) or 'all': ").strip().lower()
    if choice == "all":
        selected = active
    else:
        try:
            ids = {int(value.strip()) for value in choice.split(",")}
            selected = [item for item in active if item["id"] in ids]
            if not selected or len(selected) != len(ids):
                raise ValueError
        except ValueError:
            print("Invalid selection. Nothing deleted.")
            return

    confirm = input(
        f"Type DELETE to permanently delete {len(selected)} file(s): "
    ).strip()
    if confirm != "DELETE":
        print("Cancelled.")
        return

    for item in selected:
        stored = Path(item["stored"])
        try:
            if not is_inside(stored, REVIEW) or stored == MANIFEST:
                print(f"Skipped unsafe path: {stored}")
                continue
            if stored.is_file() and not stored.is_symlink():
                stored.unlink()
                item["status"] = "deleted"
                print(f"Deleted: {stored.name}")
            else:
                print(f"Not found or not a regular file: {stored}")
        except OSError as exc:
            print(f"Could not delete {stored}: {exc}")

    save_manifest(manifest)


def main():
    parser = argparse.ArgumentParser(
        description="Scan, review, restore, and selectively delete files."
    )
    parser.add_argument(
        "command",
        choices=("scan", "review", "restore", "purge"),
        help="operation to perform",
    )
    args = parser.parse_args()

    try:
        if args.command == "scan":
            scan()
        elif args.command == "review":
            review()
        elif args.command == "restore":
            restore()
        elif args.command == "purge":
            purge()
    except (OSError, RuntimeError) as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
